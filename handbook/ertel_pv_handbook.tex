\section{Ertel PV in Height Coordinates}

Ertel's potential vorticity \citep{ertel1942} in height coordinates $(x,y,z)$:

\begin{equation}\label{eq:epv_z}
\mathrm{EPV} = \frac{1}{\rho}\,(\vec{\omega}_a \cdot \nabla\theta)
\end{equation}

where $\vec{\omega}_a = \nabla \times \vec{u} + 2\vec{\Omega}$ (absolute
vorticity), $\theta = T(p_0/p)^{R_d/c_p}$ (dry potential temperature).

Horizontal derivatives use spectrally accurate spherical harmonic transforms
via \texttt{pvtend.sh\_ops}, eliminating polar singularities.

\section{Isobaric Formulation (Reference)}

Under hydrostatic approximation and neglecting horizontal gradients of $w$:

\begin{equation}\label{eq:epv_p}
\boxed{\mathrm{PV}_p = -g\left[
  \frac{\partial v}{\partial p}\frac{\partial\theta}{\partial x}\Big|_p
  - \frac{\partial u}{\partial p}\frac{\partial\theta}{\partial y}\Big|_p
  + (f + \zeta_p)\frac{\partial\theta}{\partial p}
\right]}
\end{equation}

\textbf{Simplified} (ERA5-like): $\mathrm{PV}_{p,\mathrm{simple}} = -g\,(f + \zeta_p)\,\partial\theta/\partial p$.
In PVU: $\mathrm{PV}_{\mathrm{PVU}} = \mathrm{PV}_{\mathrm{SI}} \times 10^6$.

\section{Sigma-Coordinate Formulation (Primary)}\label{sec:sigma}

Terrain-following sigma: $\sigma = p/p_s$ where $p_s(x,y)$ is surface
pressure.  Mimics MPAS-Atmosphere's native vertical coordinate.

Coordinate transformation $\partial/\partial p \to (1/p_s)\partial/\partial\sigma$:

\begin{equation}\label{eq:epv_sigma}
\boxed{\mathrm{PV}_\sigma = -\frac{g}{p_s}\left[
  \frac{\partial v}{\partial\sigma}\frac{\partial\theta}{\partial x}\Big|_\sigma
  - \frac{\partial u}{\partial\sigma}\frac{\partial\theta}{\partial y}\Big|_\sigma
  + (f + \zeta_\sigma)\frac{\partial\theta}{\partial\sigma}
\right]}
\end{equation}

\textbf{Analytic equivalence:} Ertel PV is coordinate-invariant:
$\mathrm{PV}_\sigma(x,y,\sigma) = \mathrm{PV}_p(x,y,p=\sigma p_s)$.

\textbf{Advantages over isobaric:}
\begin{itemize}
  \item Eliminates near-surface coordinate mismatch (850\,hPa problem).
  \item Direct MPAS native-level analog.
  \item Bypasses hybrid sigma-pressure coefficients; only $p_s$ needed.
\end{itemize}

\textbf{Simplified} form: $\mathrm{PV}_{\sigma,\mathrm{simple}} =
  -(g/p_s)\,(f + \zeta_\sigma)\,\partial\theta/\partial\sigma$.

\section{Numerical Discretization}

\subsection{Vertical Derivatives}

MPAS-style centred/sided differences. Coordinate-array ordering is
auto-detected (ascending or descending).  Boundary derivatives follow
MPAS \texttt{calc\_vertDeriv\_one}.

\subsection{Sigma Interpolation}

Fields are interpolated from pressure to sigma levels via $\log p$-linear
interpolation.  Values where $\sigma p_s$ falls outside
$[p_{\min}, p_{\max}]$ are filled from the nearest valid level.

\subsection{37-Level Sigma Grid}

Default sigma grid mirrors ERA5 pressure levels normalized by
$p_0 = 100{,}000$\,Pa:

\begin{equation}
\sigma_k \in \{1.000, 0.975, 0.950, \dots, 0.003, 0.002, 0.001\}
\end{equation}

At sea level $p \approx \sigma_k \cdot 101325$\,Pa; terrain-following
via $p = \sigma_k \cdot p_s(x,y)$.

\section{Validation}

\subsection{ERA5 (2025-01-08 00Z, 37 $\sigma$ levels)}

\begin{center}
\begin{tabular}{lcc}
\toprule
Level & RMSE (PVU) & Correlation \\
\midrule
\multicolumn{3}{c}{\textbf{Isobaric (reference)}} \\
850\,hPa & 3.53 & 0.28 \\
500\,hPa & 0.22 & 0.96 \\
250\,hPa & 0.99 & 0.97 \\
\midrule
\multicolumn{3}{c}{\textbf{Sigma-coordinate (primary)}} \\
$\sigma = 0.850$ (nom.\ 861\,hPa) & \textbf{0.19} & \textbf{0.97} \\
$\sigma = 0.500$ (nom.\ 507\,hPa) & 0.30 & 0.95 \\
$\sigma = 0.250$ (nom.\ 253\,hPa) & 1.03 & 0.98 \\
\bottomrule
\end{tabular}
\end{center}

\textbf{Key result:} Sigma coordinates fix the near-surface PV problem.
850\,hPa-level correlation improves from 0.28 to 0.97; RMSE drops from
3.53 to 0.19\,PVU.
